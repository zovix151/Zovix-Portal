def render_world_class_blueprint(blueprint_data, category, prompt):
    """Render professional blueprint with category-specific styling"""
    
    styles = {
        "robotics": {
            "bg": (8, 12, 28), "grid": (25, 45, 80), "lines": (80, 200, 255),
            "accent": (255, 100, 80), "text": (150, 210, 255), "title": (100, 220, 255),
            "shadow": (40, 80, 120), "badge": "🤖", "width": 1600, "height": 1100
        },
        "electronics": {
            "bg": (12, 18, 30), "grid": (30, 50, 80), "lines": (0, 200, 200),
            "accent": (255, 200, 0), "text": (150, 220, 220), "title": (100, 255, 200),
            "shadow": (40, 80, 120), "badge": "💻", "width": 1600, "height": 1100
        },
        "architecture": {
            "bg": (240, 242, 255), "grid": (200, 210, 230), "lines": (30, 60, 150),
            "accent": (200, 50, 50), "text": (60, 80, 160), "title": (30, 60, 150),
            "shadow": (180, 190, 210), "badge": "🏠", "width": 1600, "height": 1100
        },
        "automotive": {
            "bg": (15, 20, 30), "grid": (30, 50, 80), "lines": (200, 180, 80),
            "accent": (255, 150, 0), "text": (200, 200, 150), "title": (255, 200, 100),
            "shadow": (40, 60, 80), "badge": "🚗", "width": 1600, "height": 1100
        },
        "aerospace": {
            "bg": (10, 15, 35), "grid": (25, 45, 80), "lines": (100, 200, 255),
            "accent": (255, 255, 100), "text": (150, 200, 255), "title": (100, 220, 255),
            "shadow": (40, 60, 100), "badge": "🚀", "width": 1600, "height": 1100
        },
        "industrial": {
            "bg": (20, 25, 35), "grid": (40, 60, 80), "lines": (255, 180, 50),
            "accent": (255, 100, 50), "text": (200, 200, 150), "title": (255, 200, 80),
            "shadow": (50, 70, 90), "badge": "🏭", "width": 1600, "height": 1100
        },
        "interior": {
            "bg": (250, 248, 245), "grid": (220, 215, 210), "lines": (80, 60, 100),
            "accent": (180, 80, 120), "text": (100, 80, 120), "title": (80, 60, 100),
            "shadow": (200, 190, 200), "badge": "🛋️", "width": 1600, "height": 1100
        },
        "landscape": {
            "bg": (230, 245, 230), "grid": (190, 215, 190), "lines": (40, 120, 60),
            "accent": (200, 150, 50), "text": (60, 140, 80), "title": (40, 120, 60),
            "shadow": (180, 210, 180), "badge": "🌿", "width": 1600, "height": 1100
        },
        "futuristic": {
            "bg": (5, 5, 20), "grid": (20, 20, 60), "lines": (0, 200, 255),
            "accent": (255, 0, 200), "text": (150, 200, 255), "title": (0, 255, 255),
            "shadow": (30, 30, 80), "badge": "🌌", "width": 1600, "height": 1100
        },
        "medical": {
            "bg": (245, 248, 255), "grid": (210, 220, 240), "lines": (30, 100, 180),
            "accent": (0, 180, 100), "text": (60, 100, 160), "title": (30, 80, 160),
            "shadow": (190, 200, 220), "badge": "🏥", "width": 1600, "height": 1100
        },
        "educational": {
            "bg": (250, 248, 240), "grid": (215, 210, 200), "lines": (100, 80, 60),
            "accent": (200, 150, 50), "text": (100, 80, 60), "title": (80, 60, 40),
            "shadow": (200, 195, 185), "badge": "🎓", "width": 1600, "height": 1100
        }
    }
    
    style = styles.get(category, styles["architecture"])
    
    width = style["width"]
    height = style["height"]
    
    img = Image.new("RGB", (width, height), color=style["bg"])
    draw = ImageDraw.Draw(img)
    
    # Professional Grid
    for x in range(40, width-40, 40):
        draw.line([(x, 60), (x, height-40)], fill=style["grid"], width=1)
    for y in range(60, height-40, 40):
        draw.line([(40, y), (width-40, y)], fill=style["grid"], width=1)
    
    for x in range(40, width-40, 10):
        draw.line([(x, 60), (x, height-40)], fill=(style["grid"][0]//2, style["grid"][1]//2, style["grid"][2]//2), width=1)
    for y in range(60, height-40, 10):
        draw.line([(40, y), (width-40, y)], fill=(style["grid"][0]//2, style["grid"][1]//2, style["grid"][2]//2), width=1)
    
    draw.rectangle([(20, 20), (width-20, height-20)], outline=style["lines"], width=3)
    draw.rectangle([(25, 25), (width-25, height-25)], outline=style["shadow"], width=1)
    
    # Header
    title = blueprint_data.get('title', 'Technical Blueprint')
    description = blueprint_data.get('description', '')[:120]
    
    draw.text((width//2 - 300, 30), title, fill=style["title"], font=None)
    draw.line([(width//2 - 300, 55), (width//2 + 300, 55)], fill=style["accent"], width=2)
    draw.text((width//2 - 350, 65), description, fill=style["text"], font=None)
    
    badge = style["badge"]
    draw.text((width-160, 30), f"{badge} {category.upper()}", fill=style["accent"], font=None)
    draw.text((width-200, 55), datetime.now().strftime("%d %b %Y"), fill=style["text"], font=None)
    
    # Category-specific drawing
    if category == "robotics":
        draw_robotics_blueprint_v2(draw, style, width, height, blueprint_data)
    elif category == "electronics":
        draw_electronics_blueprint_v2(draw, style, width, height, blueprint_data)
    elif category == "architecture":
        draw_architecture_blueprint_v2(draw, style, width, height, blueprint_data)
    elif category == "automotive":
        draw_automotive_blueprint_v2(draw, style, width, height, blueprint_data)
    elif category == "aerospace":
        draw_aerospace_blueprint_v2(draw, style, width, height, blueprint_data)
    elif category == "industrial":
        draw_industrial_blueprint_v2(draw, style, width, height, blueprint_data)
    elif category == "interior":
        draw_interior_blueprint_v2(draw, style, width, height, blueprint_data)
    elif category == "landscape":
        draw_landscape_blueprint_v2(draw, style, width, height, blueprint_data)
    elif category == "futuristic":
        draw_futuristic_blueprint_v2(draw, style, width, height, blueprint_data)
    elif category == "medical":
        draw_medical_blueprint(draw, style, width, height, blueprint_data)
    elif category == "educational":
        draw_educational_blueprint(draw, style, width, height, blueprint_data)
    else:
        draw_architecture_blueprint_v2(draw, style, width, height, blueprint_data)
    
    # Specifications & Legend
    legend_x = width - 340
    legend_y = height - 220
    legend_w = 300
    legend_h = 190
    
    draw.rectangle([(legend_x, legend_y), (legend_x + legend_w, legend_y + legend_h)], 
                   fill=(style["bg"][0]+25, style["bg"][1]+25, style["bg"][2]+25), 
                   outline=style["lines"], width=2)
    
    draw.text((legend_x + 15, legend_y + 12), "📋 TECHNICAL SPECIFICATIONS", fill=style["title"], font=None)
    draw.line([(legend_x + 15, legend_y + 30), (legend_x + legend_w - 15, legend_y + 30)], fill=style["shadow"], width=1)
    
    specs = blueprint_data.get('specifications', '')
    specs_lines = specs.split('|') if specs else []
    y_pos = legend_y + 45
    for line in specs_lines[:4]:
        draw.text((legend_x + 15, y_pos), "▸ " + line.strip(), fill=style["text"], font=None)
        y_pos += 22
    
    dims = blueprint_data.get('dimensions', '')
    if dims:
        draw.text((legend_x + 15, y_pos + 10), f"📐 {dims}", fill=style["accent"], font=None)
    
    # Footer
    draw.text((50, height-25), f"© ZOVIX AI • {datetime.now().strftime('%Y')} • {category.upper()} ENGINEERING", 
              fill=style["text"], font=None)
    draw.text((width-300, height-25), "Generated by ZOVIX World Class Engine v2.0", 
              fill=(style["text"][0]//2, style["text"][1]//2, style["text"][2]//2), font=None)
    
    output_path = f"blueprints/blueprint_{uuid.uuid4().hex[:8]}.png"
    os.makedirs("blueprints", exist_ok=True)
    img.save(output_path, quality=95)
    
    return output_path
